cmake_minimum_required(VERSION 3.20)
project(projectz)
set(CMAKE_CXX_STANDARD 17)
set(SFML_DIR "C:/sfml/lib/cmake/SFML")
set(SFML_STATIC_LIBRARIES TRUE)
find_package(SFML 2.6 REQUIRED COMPONENTS graphics window system)
add_executable(projectz src/main.cpp)
target_compile_definitions(projectz PRIVATE SFML_STATIC)
target_link_libraries(projectz sfml-graphics sfml-window sfml-system)
add_custom_command(TARGET projectz POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/assets $<TARGET_FILE_DIR:projectz>/assets)
